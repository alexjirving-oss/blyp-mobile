rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    
    // TikTok-style streaming rules with optimized access
    match /streams/{userId}/{streamPath=**} {
      // Public read access for all stream segments (TikTok-style)
      allow read: if true;
      
      // Write access only for authenticated stream owner
      allow write: if request.auth != null 
                   && request.auth.uid == userId
                   && request.resource.size < 50 * 1024 * 1024; // 50MB limit per segment
      
      // Delete access for cleanup (TikTok-style storage management)
      allow delete: if request.auth != null 
                    && request.auth.uid == userId;
    }
    
    // Thumbnails and profile images
    match /thumbnails/{userId}/{allPaths=**} {
      allow read: if true;
      allow write: if request.auth != null 
                   && request.auth.uid == userId
                   && request.resource.size < 10 * 1024 * 1024; // 10MB limit
    }
  }
}
